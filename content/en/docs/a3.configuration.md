## Add Configuration

> [!tip]
> - We use environment variables to configure the application.
> - We use [`caarlos0/env`](https://github.com/caarlos0/env/) to initialize configuration `struct` from environment variables on application start, and immediately panic if no configuration is set.


### Add `caarlos0/env/v11`

Run following `go get` command from the project root.

```shell
go get github.com/caarlos0/env/v11
```

### Add Initial `.env`

Add a new file `.env` to the project root.

```env
TZ=Asia/Singapore

SERVER_PORT=8080
SERVER_TIMEOUT_READ=3s
SERVER_TIMEOUT_WRITE=5s
SERVER_TIMEOUT_IDLE=5s
SERVER_DEBUG=true

CORS_ALLOWED_ORIGINS="http://localhost:8080,http://127.0.0.1:8080"
CORS_ALLOWED_HEADERS="Accept,Authorization,Content-Type,X-CSRF-Token"
CORS_MAX_AGE=300
```

### Add `Conf` Struct

Add a new file `config/config.go`.

```go
package config

import (
    "log"
    "time"

    "github.com/caarlos0/env/v11"
)

type Conf struct {
    TZ     string `env:"TZ,required"`
    Server ConfServer
    CORS   ConfCORS
}

type ConfServer struct {
    Port           int           `env:"SERVER_PORT,required"`
    TimeoutRead    time.Duration `env:"SERVER_TIMEOUT_READ,required"`
    TimeoutWrite   time.Duration `env:"SERVER_TIMEOUT_WRITE,required"`
    TimeoutIdle    time.Duration `env:"SERVER_TIMEOUT_IDLE,required"`
    Debug          bool          `env:"SERVER_DEBUG,required"`
}

type ConfCORS struct {
    AllowedOrigins string `env:"CORS_ALLOWED_ORIGINS,required"`
    AllowedHeaders string `env:"CORS_ALLOWED_HEADERS,required"`
    MaxAge         int    `env:"CORS_MAX_AGE,required"`
}

func New() *Conf {
    var c Conf
    if err := env.Parse(&c); err != nil {
        log.Fatalf("Failed to decode: %s", err)
    }

    return &c
}
```

Update `cmd/app/main.go` to use new `Conf struct`.

```go
package main

import (
    "fmt"
    "net/http"
    "os"
    "time"

    "github.com/rs/zerolog"

    "myapp/app/router"
    "myapp/config"
)

func main() {
    c := config.New()

    loc, _ := time.LoadLocation(c.TZ)
    time.Local = loc

    logLevel := zerolog.InfoLevel
    if c.Server.Debug {
        logLevel = zerolog.DebugLevel
    }
    zerolog.SetGlobalLevel(logLevel)
    l := zerolog.New(os.Stderr).With().Timestamp().Logger()

    mux := router.New(&c.CORS, &l)

    s := &http.Server{
        Addr:         fmt.Sprintf(":%d", c.Server.Port),
        Handler:      mux,
        ReadTimeout:  c.Server.TimeoutRead,
        WriteTimeout: c.Server.TimeoutWrite,
        IdleTimeout:  c.Server.TimeoutIdle,
    }

    l.Info().Msgf("Starting server %v", s.Addr)
    s.ListenAndServe()
}
```

## Update Router

### Add CORS Support

Run following `go get` command from the project root.

```shell
go get github.com/go-chi/cors 
```

Update `app/router/router.go` to add CORS middleware.

```go
package router

import (
    "net/http"
    "strings"

    "github.com/go-chi/chi/v5"
    "github.com/go-chi/cors"
    "github.com/rs/zerolog"
    "github.com/rs/zerolog/hlog"

    "myapp/app/book"
    "myapp/config"
    mrl "myapp/pkg/middleware/requestlog"
)

func New(c *config.ConfCORS, l *zerolog.Logger) *chi.Mux {
    r := chi.NewRouter()
    r.Get("/livez", func(w http.ResponseWriter, _ *http.Request) {
        w.Write([]byte("."))
    })

    cors := cors.New(cors.Options{
        AllowedOrigins:   strings.Split(c.AllowedOrigins, ","),
        AllowedMethods:   []string{http.MethodGet, http.MethodPost, http.MethodPut, http.MethodDelete, http.MethodOptions},
        AllowedHeaders:   strings.Split(c.AllowedHeaders, ","),
        AllowCredentials: true,
        ExposedHeaders:   []string{""},
        MaxAge:           c.MaxAge,
    })

    r.Route("/v1", func(r chi.Router) {
        r.Use(cors.Handler)
        r.Use(hlog.NewHandler(*l))
        r.Use(hlog.RequestIDHandler("request_id", "X-Request-ID"))
        r.Use(mrl.NewHandler)

        r.Route("/books", book.New().Register)
    })

    return r
}
```

> [!tip] 
> Run `go mod tidy` every time after adding changes with a new dependency.

### Add `application/json` Content-Type Middleware

Add new files `content_type.go` and `content_type_test.go` to `pkg/middleware` directory.

```go {title="pkg/middleware/content_type.go"}
package middleware

import "net/http"

const (
    HeaderKeyContentType       = "Content-Type"
    HeaderValueContentTypeJSON = "application/json;charset=utf8"
)

func ContentTypeJSON(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        w.Header().Set(HeaderKeyContentType, HeaderValueContentTypeJSON)
        next.ServeHTTP(w, r)
    })
}
```

```go {title="pkg/middleware/content_type_test.go"}
package middleware_test

import (
    "fmt"
    "net/http"
    "net/http/httptest"
    "testing"

    "myapp/pkg/middleware"
)

const testRespBody = `{"k":"v"}`

func TestContentTypeJSON(t *testing.T) {
    r, _ := http.NewRequest("GET", "/", nil)
    w := httptest.NewRecorder()

    middleware.ContentTypeJSON(http.HandlerFunc(testHandlerFunc())).ServeHTTP(w, r)
    response := w.Result()

    if respBody := w.Body.String(); respBody != testRespBody {
        t.Errorf("Wrong response body:  got %v want %v ", respBody, testRespBody)
    }

    if status := response.StatusCode; status != http.StatusOK {
        t.Errorf("Wrong status code: got %v want %v", status, http.StatusOK)
    }

    if contentType := response.Header.Get(middleware.HeaderKeyContentType); contentType != middleware.HeaderValueContentTypeJSON {
        t.Errorf("Wrong content-type: got %v want %v", contentType, middleware.HeaderValueContentTypeJSON)
    }
}

func testHandlerFunc() func(w http.ResponseWriter, r *http.Request) {
    return func(w http.ResponseWriter, r *http.Request) {
        fmt.Fprint(w, testRespBody)
    }
}
```
