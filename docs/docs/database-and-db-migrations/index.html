Database and DB Migrations

Database Design

Tip
  • To keep this series simple, we use only a single Postgres database table named books.
  • We order fixed-width table columns by decreasing alignment requirements to optimize using Column Tetris.
  • However, it’s ok to follow a more readable column format, when your table schema changes frequently.
Column NameDatatypeNot NullPrimary Key
created_atTIMESTAMPTZ✅
updated_atTIMESTAMPTZ✅
idUUID✅✅
published_dateDATE✅
statusSMALLINT✅
titleTEXT✅
descriptionTEXT
image_urlTEXT

Add Initial compose.yml

Add a new file compose.yml to the project root.

services:
  db:
    image: postgres:18-alpine
    environment:
      TZ: "Asia/Singapore"
      POSTGRES_DB: book-service-db
      POSTGRES_USER: book-service-user
      POSTGRES_PASSWORD: book-service-pass
    ports:
      - "5432:5432"
    healthcheck:
      test: [ "CMD-SHELL", "pg_isready -U book-service-user -d book-service-db" ]
      interval: 3s
      timeout: 5s
      retries: 5
    restart: always

Update Conf Struct

Update config/config.go.

// ...
type Conf struct {
	TZ     string `env:"TZ,required"`
	Server ConfServer
	CORS   ConfCORS
	DB     ConfDB
}

// ... ConfServer, ConfCORS

type ConfDB struct {
    Host     string `env:"DB_HOST,required"`
    Port     int    `env:"DB_PORT,required"`
    Username string `env:"DB_USER,required"`
    Password string `env:"DB_PASS,required"`
    DBName   string `env:"DB_NAME,required"`
    Debug    bool   `env:"DB_DEBUG,required"`
}

// ... func New

func NewDB() *ConfDB {
    var c ConfDB
    if err := env.Parse(&c); err != nil {
        log.Fatalf("Failed to decode: %s", err)
    }

    return &c
}

Add Migration CLI

Tip
  • We use goose as a library to build a migration CLI with embeded SQLs.
  • We use pgx as the PostgreSQL driver and toolkit.

Add Goose and PGX

Run following go get commands from the project root.

go get github.com/pressly/goose/v3
go get github.com/jackc/pgx/v5

Add Migration CLI

Run following commands from the project root.

mkdir -p cmd/migration/migrations && touch cmd/migration/{main.go,usage_embed.go,usage_no_embed.go}

These commands will generate the following folder structure,

myapp
└── cmd
    └── migration
        ├── main.go
        ├── usage_embed.go
        ├── usage_no_embed.go
        └── migrations/

Update main.go, usage_embed.go and usage_no_embed.go files in cmd/migration directory.

main.go
cmd/migration/main.go
package main

import (
    "context"
    "flag"
    "fmt"
    "log"
    "os"

    _ "github.com/jackc/pgx/v5/stdlib"
    "github.com/pressly/goose/v3"

    "myapp/config"
)

const (
    dialect     = "pgx"
    fmtDBString = "host=%s user=%s password=%s dbname=%s port=%d sslmode=disable"
)

var (
    flags = flag.NewFlagSet("migration", flag.ExitOnError)
    dir   = flags.String("dir", "migrations", "directory with migration files")
)

func main() {
    flags.Usage = usage
    flags.Parse(os.Args[1:])

    args := flags.Args()
    if len(args) == 0 || args[0] == "-h" || args[0] == "--help" {
        flags.Usage()
        return
    }

    command := args[0]

    c := config.NewDB()
    dbString := fmt.Sprintf(fmtDBString, c.Host, c.Username, c.Password, c.DBName, c.Port)

    db, err := goose.OpenDBWithDriver(dialect, dbString)
    if err != nil {
        log.Fatal(err.Error())
    }

    defer func() {
        if err := db.Close(); err != nil {
            log.Fatal(err.Error())
        }
    }()

    if err := goose.RunContext(context.Background(), command, db, *dir, args[1:]...); err != nil {
        log.Fatalf("migration %v: %v", command, err)
    }
}
usage_embed.go
cmd/migration/usage_embed.go
//go:build embed

package main

import (
    "embed"
    "fmt"

    "github.com/pressly/goose/v3"
)

//go:embed migrations/*.sql
var migrations embed.FS

func init() {
    goose.SetBaseFS(migrations)
}

func usage() {
    fmt.Println(usageCommands)
}

var usageCommands = `Usage: migration COMMAND
Examples:
  migration status

Commands:
  up                   Migrate the DB to the most recent version available
  up-by-one            Migrate the DB up by 1
  up-to VERSION        Migrate the DB to a specific VERSION
  down                 Roll back the version by 1
  down-to VERSION      Roll back to a specific VERSION
  redo                 Re-run the latest migration
  reset                Roll back all migrations
  status               Dump the migration status for the current DB
  version              Print the current version of the database`

# List available commands
help:
    @just --list --unsorted --list-heading $'MYAPP\n'

# Run server app
app:
    @export $(grep -v '^#' .env | xargs) && \

migration *cmd="up":

build:
    @docker compose build

# Run docker compose up
up cmd="":
    @docker compose up {{ cmd }}

# Run docker compose down
down:
    @docker compose down

migration commands

Usage: migration COMMAND
Examples:
  migration create initial_tables sql

  -dir string
        directory with migration files (default "migrations")

Commands:
  create NAME [sql|go] Creates new migration file with the current timestamp
  fix                  Apply sequential ordering to migrations
  up                   Migrate the DB to the most recent version available
  up-by-one            Migrate the DB up by 1
  up-to VERSION        Migrate the DB to a specific VERSION
  down                 Roll back the version by 1
  down-to VERSION      Roll back to a specific VERSION
  redo                 Re-run the latest migration
  reset                Roll back all migrations
  status               Dump the migration status for the current DB
  version              Print the current version of the database

create

Example: just migration create create_books_table sql creates a new SQL file in the migrations directory.

Created new file: myapp/cmd/migration/migrations/20270101000000_create_books_table.sql

fix

Example: just migration fix

RENAMED 20270101000000_create_books_table.sql => 00001_create_books_table.sql

status

Example: just migration status

Applied At                  Migration
=======================================
Pending                  -- 00001_create_books_table.sql

up

Example: just migration up

OK   00001_create_books_table.sql (3.49ms)
goose: successfully migrated database to version: 1

down

Example: just migration down

OK   00001_create_books_table.sql (2.64ms)

version

Example: just migration version

goose: version 0

Add Initial Migration SQL

Update cmd/migration/migrations/00001_create_books_table.sql.

-- +goose Up
-- SQL in this section is executed when the migration is applied.
CREATE TABLE IF NOT EXISTS books
(
    created_at     TIMESTAMPTZ NOT NULL,
    updated_at     TIMESTAMPTZ NOT NULL,
    id             UUID        NOT NULL,
    published_date DATE        NOT NULL,
    status         SMALLINT    NOT NULL,
    title          TEXT        NOT NULL,
    description    TEXT,
    image_url      TEXT,
    PRIMARY KEY (id)
);

-- +goose Down
-- SQL in this section is executed when the migration is rolled back.
DROP TABLE IF EXISTS books;

Run just migration up to run the migration file.
